#include <unistd.h>
#include <fcntl.h>
#include <cstring>
#include <sys/wait.h>
#include <sys/mman.h>
#include <semaphore.h>
#include <cstdio>
#include <cstdlib>

#define MAX_NUMS 10000
#define MAX_LINES 1000

struct SharedData {
    double input_nums[MAX_NUMS];
    int line_lengths[MAX_LINES];
    int total_lines;

    double output_result[MAX_LINES];
    int error_flag;
};

int parse_line(char* line, double* nums) {
    int count = 0;
    char* p = line;
    char* end;
    while (true) {
        double x = strtod(p, &end);
        if (end == p) break;
        nums[count++] = x;
        p = end;
    }
    return count;
}

int main() {
    char file_name[256];

    ssize_t n = read(0, file_name, sizeof(file_name) - 1);

    if (n <= 0) return 1;

    file_name[n] = '\0';
    if (file_name[n - 1] == '\n') file_name[n - 1] = '\0';

    int file = open(file_name, O_RDONLY);
    if (file == -1) {
        const char* msg = "open file error\n";
        write(2, msg, strlen(msg));
        return 1;
    }

    char shm_name[64], sem_data_name[64], sem_result_name[64];
    pid_t pid = getpid();
    snprintf(shm_name, sizeof(shm_name), "/shm_%d", pid);
    snprintf(sem_data_name, sizeof(sem_data_name), "/sem_d_%d", pid);
    snprintf(sem_result_name, sizeof(sem_result_name), "/sem_r_%d", pid);

    int shm_fd = shm_open(shm_name, O_CREAT | O_RDWR, 0666);
    ftruncate(shm_fd, sizeof(SharedData));
    SharedData* shared = (SharedData*)mmap(NULL, sizeof(SharedData), PROT_READ | PROT_WRITE, MAP_SHARED, shm_fd, 0);

    sem_t* sem_data = sem_open(sem_data_name, O_CREAT, 0666, 0);
    sem_t* sem_result = sem_open(sem_result_name, O_CREAT, 0666, 0);

    char file_buf[8192];
    ssize_t r = read(file, file_buf, sizeof(file_buf) - 1);
    if (r > 0) file_buf[r] = '\0';
    close(file);

    shared->total_lines = 0;
    int num_index = 0;

    char* line = file_buf;
    while (line != nullptr && *line != '\0') {
        char* nl = strchr(line, '\n');
        if (nl != nullptr) *nl = '\0';

        int count = parse_line(line, &shared->input_nums[num_index]);
        if (count > 0) {
            shared->line_lengths[shared->total_lines] = count;
            num_index += count;
            shared->total_lines++;
        }

        if (nl == nullptr) break;
        line = nl + 1;
    }

    pid = fork();

    if (pid == -1) {
        const char* msg = "fork error\n";
        write(2, msg, strlen(msg));
        return 1;
    }

    if (pid == 0){

        sem_wait(sem_data);
        shared->error_flag = 0;

        int idx = 0;
        for (int l = 0; l < shared->total_lines; l++) {
            int len = shared->line_lengths[l];
            if (len == 0) continue;
            double result = shared->input_nums[idx];
            for (int i = 1; i < len; i++) {
                double x = shared->input_nums[idx + i];
                if (x == 0.0) {
                    shared->error_flag = 1;
                    sem_post(sem_result);
                    exit(1);
                }
                result /= x;
            }
            shared->output_result[l] = result;
            idx += len;
        }
        sem_post(sem_result);
        exit(0);

    } else {

        sem_post(sem_data);
        sem_wait(sem_result);

        int status;
        waitpid(pid, &status, 0);

        if (shared->error_flag == 1) {
            const char* msg = "division by zero\n";
            write(2, msg, strlen(msg));
        } else {

            for (int l = 0; l < shared->total_lines; l++) {
                char out[64];
                int len = snprintf(out, sizeof(out), "%.4f\n", shared->output_result[l]);
                write(1, out, len);
            }
        }

        munmap(shared, sizeof(SharedData));
        close(shm_fd);
        shm_unlink(shm_name);

        sem_close(sem_data);
        sem_unlink(sem_data_name);

        sem_close(sem_result);
        sem_unlink(sem_result_name);

    }

    return 0;
}
